#Using String data structures

import re
string = input("Enter String: ")
#only alphabetical values

if re.fullmatch("[a-zA-Z0-9]+", string):
    print("Valid string")
else:
    print("Invalid string")

Output:
Enter String:  ishi 
Valid string
