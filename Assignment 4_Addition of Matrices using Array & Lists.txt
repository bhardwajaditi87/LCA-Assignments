#Addition of Matrices using Lists and Array

import numpy as np

a = np.array([[1, 2], [3, 4]])
b = np.array([[5, 6], [7, 8]])
c = a + b

print("Matrix A:")
print(a)

print("Matrix B:")
print(b)

print("Addition of matrices:")
print(c)

Output:
Matrix A:
[[1 2]
 [3 4]]
Matrix B:
[[5 6]
 [7 8]]
Addition of matrices:
[[ 6  8]
 [10 12]]
